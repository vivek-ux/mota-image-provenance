\begin{frame}{Motivation: What is Composite Media Lineage?}
  \begin{itemize}\setlength{\itemsep}{5pt}
    \item \textbf{Single-Source vs. Composite Derivatives:} Real-world visual tampering is rarely 1-to-1 duplication. Forensic audits confront \textbf{multi-source composites}:
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item \textbf{Parent A (Background Plate):} Occupies $70\%\text{--}85\%$ of the total canvas.
      \item \textbf{Parent B (Spliced Donor Object):} Localized transfer occupying only $10\%\text{--}30\%$.
    \end{itemize}
    \item \textbf{High-Impact Real-World Threat Scenarios:}
    \begin{itemize}\setlength{\itemsep}{3pt}
      \item \textbf{Evidentiary Contraband Planting:} Weapons or narcotics spliced into crime scene evidence tables.
      \item \textbf{Vehicle Registration Fraud:} High-Security Registration Plates (HSRP) forged onto hit-and-run cars.
      \item \textbf{Alibi Fabrication:} Splicing human subjects into scenic overlooks to fabricate geographic presence.
    \end{itemize}
    \item \textbf{The Core Objective:} In an open-world gallery of $N$ authenticated assets, accurately retrieve \textbf{both Parent A and Parent B} and prove which arrived first.
  \end{itemize}
\end{frame}

% ==============================================================================
% SLIDE 3: THE CORE PROBLEM: BACKGROUND DOMINANCE PARADOX (45 Seconds)
% ==============================================================================
\begin{frame}{The Background Dominance Paradox}
  \begin{block}{Why State-of-the-Art Deep Vision Models Fail on Composites}
    Standard copy-detection models (Meta SSCD, ResNet-50, CLIP) condense an entire image into a single global vector $\mathbf{v}_{\text{global}} \in \mathbb{R}^D$ via Global Average Pooling (GAP).
  \end{block}
  \vspace{0.15cm}
  \begin{itemize}\setlength{\itemsep}{4pt}
    \item \textbf{The Mathematical Flaw:} 
    Because Parent A dominates the pixel space ($\ge 75\%$), global cosine similarity is overwhelmed by background features:
    \[
    \cos(\mathbf{v}_{\text{composite}}, \mathbf{v}_{\text{Parent A}}) \approx 0.96 \quad \text{vs.} \quad \cos(\mathbf{v}_{\text{composite}}, \mathbf{v}_{\text{Parent B}}) \le 0.42
    \]
    \item \textbf{Result:} SOTA baseline models achieve \textbf{100\% Parent A recall} but \textbf{0\% dual-parent recovery}. Parent B is lost in background noise.
    \item \textbf{Why Rigid Grids Fail:} Dividing the image into fixed $2\times2$ or $3\times3$ non-overlapping tiles chops central donor objects into four pieces, destroying semantic integrity.
  \end{itemize}
\end{frame}

\begin{frame}{Proposed Architecture: Two-Stream Neural Patch Consensus}
  \begin{block}{1. Multi-Scale Dynamic Patch Tokenizer}
    \small Instead of rigid square grids, we extract multi-scale overlapping windows at $25\%$, $35\%$, and $50\%$ canvas dimensions with a \textbf{$50\%$ stride overlap}. This mathematically guarantees any donor is captured intact.
  \end{block}
  \vspace{0.1cm}
  \begin{block}{2. Learned MLP Saliency Importance Gate}
    \small Flat skies and blank walls add noise. Our 2-layer MLP computes an attention weight $w_i \in [0, 1]$ based on spatial feature variance:
    \[
    w_i = \sigma\big(\mathbf{W}_2 \cdot \text{ReLU}(\mathbf{W}_1 \mathbf{f}_i + \mathbf{b}_1) + \mathbf{b}_2\big)
    \]
  \end{block}
  \vspace{0.1cm}
  \begin{block}{3. Max-Consensus Lineage Aggregator ($\alpha = 0.25$)}
    \small Combines global background score with gated localized patch maximums:
    \[
    \mathcal{S}_{\text{lineage}}(Q, G_j) = 0.25 \cdot \mathcal{S}_{\text{global}}(Q, G_j) + 0.75 \cdot \max_{i} \big( w_i \cdot \cos(\mathbf{f}_{Q,i}, \mathbf{f}_{G_j}) \big)
    \]
  \end{block}
\end{frame}

  \begin{block}{Why AI Similarity Alone is Insufficient for Evidentiary Law}
    Deep learning predicts similarity percentages, but \textbf{cannot establish temporal precedence}. A counterfeiter can steal an original image and claim their forged composite came first.
  \end{block}
  \vspace{0.12cm}
  \begin{itemize}\setlength{\itemsep}{4pt}
    \item \textbf{Decentralized SHA-256 Ledger:} Every original photograph and composite is cryptographically digested and minted into an immutable block with Proof-of-Work nonce.
    \item \textbf{Live On-Chain State:} \textbf{129 verified blocks} mined and operational in our live system (`blockchain.py`).
    \item \textbf{Mathematical Temporal Precedence Theorem:}
    \[
    \text{Certified Composite}(Q) \iff \mathcal{T}(\text{Parent A}) < \mathcal{T}(Q) \;\land\; \mathcal{T}(\text{Parent B}) < \mathcal{T}(Q)
    \]
    \item Generates cryptographically verifiable receipts admissible in copyright and forensic audits.
  \end{itemize}
\end{frame}

\begin{frame}{Failure Mode Analysis: Latent Diffusion \& Spectral Gating}
  \begin{block}{Where Physical Retrieval Fails: Generative Latent Inpainting}
    When an image is modified using generative diffusion (e.g., Meta AI Emu, Stable Diffusion), \textbf{no physical donor exists in the real world}.
  \end{block}
  \vspace{0.1cm}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item \textbf{The Failure Mode:} Unguided nearest-neighbor search forced false-positive matches (matching a generated sun to yellow headphones based on color).
    \item \textbf{Root Cause:} Diffusion models synthesize pixels in latent space $\mathcal{Z}$, lacking camera sensor PRNU noise patterns.
    \item \textbf{Our Solution: 2D FFT Spectral Fourier Gating:}
    \begin{itemize}\setlength{\itemsep}{2pt}
      \item Analyzes radial high-frequency power spectrum decay $E_{\text{HF}}$.
      \item \textbf{Null-Hypothesis Rejection Gate:} If $E_{\text{HF}} < \tau_{\text{spectral}}$, the system refuses to guess a physical donor and flags: \textcolor{myberry}{\textbf{Synthetic Latent Inpainting (No Physical Donor)}}.
    \end{itemize}
    \item Parent A is still accurately attributed to the original on-chain background plate!
  \end{itemize}
\end{frame}
